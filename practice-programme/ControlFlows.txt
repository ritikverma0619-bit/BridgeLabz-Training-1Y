//for loop

public class ControlFlows{
public static void main (String[] args){
for(int p=1; p<=5; p++){
for(int k=1; k<=5; k++){
System.out.println("*");
}
System.out.println();
}

//while loop

int i=1;
while(i<=20){
System.out.println(i+ "");
i++;
}

//do while loop

do{
System.out.println(j);
j++;
}
while(j<=20);
}
}
